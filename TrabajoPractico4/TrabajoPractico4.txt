// Ejercicio 1:

-- REQUISITOS:

Modelar un sistema donde: - Cada Empleado pertenece a un Departamento . 
- Los Proyectos pueden tener asignados múltiples empleados con horas semanales. 
- Un empleado puede liderar uno o varios proyectos.

-- ACTIVIDADES:

    Modelado - Crear nodos y relaciones con ejemplos de datos:
        Al menos 3 empleados en diferentes departamentos.
        2 proyectos.
        Asignaciones de horas semanales.
        Un líder por proyecto.

    Consultas:
    - Obtener el nombre del proyecto, su líder y los empleados asignados. }
    - Calcular el total de horas semanales por proyecto. 
    - Listar los empleados que trabajan en más de un proyecto.

-- MODELADO:

CREATE
(Sistemas:Departamento {nombre_depto:"Sistemas"}),
(RR_Humanos:Departamento {nombre_depto:"Desarrolladores"}),
(Marketing:Departamento {nombre_depto:"Testing"}),
(web_app:Proyecto {titulo_proyecto: "Aplicacion Web"}),
(rest_api:Proyecto {titulo_proyecto: "API Rest"}),
(juana:Empleado {nombre: "Juana", horas_semanales: 48}),
(juan:Empleado {nombre: "Juan", horas_semanales: 36}),
(pablo:Empleado {nombre: "Pablo", horas_semanales: 50})



MATCH ((e:Empleado) WHERE(e.nombre="Juan")), ((d:Departamento) 
WHERE (d.nombre_depto="Testing")) MERGE (e)-[:PERTENECE_A]->(d);

MATCH ((e:Empleado) WHERE(e.nombre="Juana")), ((d:Departamento) 
WHERE (d.nombre_depto="Sistemas")) MERGE (e)-[:PERTENECE_A]->(d);

MATCH ((e:Empleado) WHERE(e.nombre="Pablo")), ((d:Departamento) 
WHERE (d.nombre_depto="Desarrolladores")) MERGE (e)-[:PERTENECE_A]->(d);



MATCH ((e:Empleado) WHERE(e.nombre="Juan")), ((p:Proyecto) 
WHERE (p.titulo_proyecto="API Rest")) MERGE (e)-[:PROYECTO_ASIGNADO]->(p);

MATCH ((e:Empleado) WHERE(e.nombre="Juan")), ((p:Proyecto) 
WHERE (p.titulo_proyecto="Aplicacion Web")) MERGE (e)-[:PROYECTO_ASIGNADO]->(p);

MATCH ((e:Empleado) WHERE(e.nombre="Pablo")), ((p:Proyecto) 
WHERE (p.titulo_proyecto="Aplicacion Web")) MERGE (e)-[:PROYECTO_ASIGNADO]->(p);

MATCH ((e:Empleado) WHERE(e.nombre="Juana")), ((p:Proyecto) 
WHERE (p.titulo_proyecto="API Rest")) MERGE (e)-[:PROYECTO_ASIGNADO]->(p);



MATCH((p:Proyecto) WHERE(p.titulo_proyecto="API Rest")), ((e:Empleado) WHERE (e.nombre="Juan")) 
SET p.lider=e.nombre;

MATCH((p:Proyecto) WHERE(p.titulo_proyecto="Aplicacion Web")), ((e:Empleado) WHERE (e.nombre="Pablo")) 
SET p.lider=e.nombre;



-- CONSULTAS:
MATCH(e:Empleado)-[r:PROYECTO_ASIGNADO]-(p:Proyecto) RETURN e.nombre, r, p.titulo_proyecto, p.lider;

MATCH(e:Empleado)-[:PROYECTO_ASIGNADO]->(p:Proyecto) WITH e, 
count(p) AS num_proyectos_asignados WHERE num_proyectos_asignados > 1 MATCH (e)-[:PROYECTO_ASIGNADO]->(proyecto) 
RETURN e.nombre, num_proyectos_asignados, collect(proyecto.titulo_proyecto) as proyectos;














// Ejercicio 2

-- REQUISITOS:

Modelar un sistema de préstamos donde: 
- Estudiantes piden libros prestados. 
- Cada libro pertenece a una Categoría . 
- Las relaciones de préstamo tienen fecha y estado ( "Activo" o "Devuelto" ). 
- Cada estudiante pertenece a una Carrera .

-- ACTIVIDADES:

    Modelado - Crear datos con al menos:
        3 estudiantes de distintas carreras.
        4 libros de categorías diferentes.
        5 préstamos con fechas y estado.

    Consultas:
    - Obtener todos los libros actualmente prestados (estado "Activo" ). 
    - Listar cuántos libros ha pedido prestado cada estudiante. 
    - Mostrar las categorías con más préstamos activos. 
    - Encontrar los estudiantes que no tienen préstamos activos.

-- MODELADO:

CREATE(juan:Alumno {nombre: "Juan", apellido: "Gomez"}),
(gonzalo:Alumno {nombre: "Gonzalo", apellido: "Perez"}),
(santiago:Alumno {nombre: "Santiago", apellido: "Alvarez"}),
(medicina: Carrera {titulo_carrera: "Medicina"}),
(ingenieria: Carrera {titulo_carrera: "Ingenieria"}),
(enfermeria: Carrera {titulo_carrera: "Enfermeria"}),
(harry_potter: Libro {titulo_libro: "Harry Potter", categoria: "Magia"}),
(cronicas_venado: Libro {titulo_libro: "Los dias del Venado", categoria: "Mitologia"}),
(cien_anios: Libro {titulo_libro: "100 años de soledad", categoria: "Antiguedad"}),
(clean_code: Libro {titulo_libro: "Clean code", categoria: "Programacion"});

MATCH((a:Alumno) WHERE (a.nombre="Gonzalo")), 
((l:Libro) WHERE (l.titulo_libro="Harry Potter"))
MERGE (a)-[:TOMO_PRESTADO {fecha: '25/2/2025', estado: "Activo"}]->(l);

MATCH((a:Alumno) WHERE (a.nombre="Juan")), 
((l:Libro) WHERE (l.titulo_libro="Harry Potter")) 
MERGE (a)-[:TOMO_PRESTADO {fecha: "30/6/2024", estado: "Devuelto"}]->(l);

MATCH((a:Alumno) WHERE (a.nombre="Juan")), 
((l:Libro) WHERE (l.titulo_libro="100 años de soledad")) 
MERGE (a)-[:TOMO_PRESTADO {fecha: "22/8/2025", estado: "Devuelto"}]->(l);

MATCH((a:Alumno) WHERE (a.nombre="Santiago")), 
((l:Libro) WHERE (l.titulo_libro="Los dias del Venado")) 
MERGE (a)-[:TOMO_PRESTADO {fecha: "30/11/2024", estado: "Activo"}]->(l);

MATCH((a:Alumno) WHERE (a.nombre="Santiago")), 
((l:Libro) WHERE (l.titulo_libro="Harry Potter")) 
MERGE (a)-[:TOMO_PRESTADO {fecha: "20/5/2024", estado: "Activo"}]->(l);

-- CONSULTAS:

MATCH (a:Alumno)-[r:TOMO_PRESTADO]-(l:Libro) 
WHERE r.estado="Activo" 
return a.nombre AS Nombre_Alumno, 
l.titulo_libro AS Titulo_Libro, 
r.estado AS Estado_Prestamo;

MATCH (a:Alumno)-[r:TOMO_PRESTADO]-(l:Libro) 
RETURN a.nombre as Nombre_Alumno, count(*) AS Cantidad_Libros_Pedidos

MATCH (a:Alumno)-[r:TOMO_PRESTADO]-(l:Libro) WHERE r.estado="Activo" 
RETURN l.categoria AS Categoria_Libro, count(*) AS Cantidad_Prestamos_Activos

MATCH (a:Alumno)
OPTIONAL MATCH (a)-[r:TOMO_PRESTADO]->(l:Libro)
WHERE r.estado = "Activo"
WITH a, count(r) AS Prestamos_Activos
WHERE Prestamos_Activos = 0
RETURN a.nombre, Prestamos_Activos;

Modelar un sistema académico que contemple: 
- Estudiantes inscritos en Cursos . 
- Cada curso corresponde a una Materia . 
- Los estudiantes obtienen calificaciones finales. 
- Las materias pueden tener prerrequisitos.

-- ACTIVIDADES:

Modelado - Crear datos de:

    3 estudiantes.
    3 materias (al menos una con prerrequisito).
    4 cursos dictados.
    Inscripciones y calificaciones.

Consultas 
- Listar la transcripción académica de un estudiante. 
- Verificar si un estudiante puede inscribirse en una materia (si aprobó los prerrequisitos). 
- Calcular el promedio de calificaciones por estudiante. 
- Detectar materias con promedio inferior a 7.

-- MODELADO:

CREATE (e1:Estudiante {legajo: 10897, nombre: "Ana"}),
       (e2:Estudiante {legajo: 8970, nombre: "Carlos"}),
       (e3:Estudiante {legajo: 3313, nombre: "Lucía"});
